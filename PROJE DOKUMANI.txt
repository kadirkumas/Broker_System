  import shutil
  import os

  SRC = 'backend/dosya.py'
  BAK = SRC + '.bak_ozellik_adi'

  if not os.path.exists(SRC):
      print(f"[HATA] {SRC} bulunamadi")
      exit(1)
  shutil.copy2(SRC, BAK)
  print(f"[1/4] Yedek: {BAK}")

  changes = 0

  with open(SRC, 'r', encoding='utf-8', newline='') as f:
      content = f.read().replace('\r\n', '\n')

  old = "..."
  new = "..."
  if old in content:
      content = content.replace(old, new, 1)
      changes += 1
      print(f"[2/4] Degisiklik uygulandi")
  else:
      print(f"[2/4] UYARI: hedef bulunamadi")

  with open(SRC, 'w', encoding='utf-8', newline='') as f:
      f.write(content.replace('\n', '\r\n'))

  print(f"BASARILI: {changes} degisiklik")
  print(f"Geri donmek icin: Copy-Item {BAK} {SRC} -Force")

  Cozum: max_symbols=30, scan_interval_seconds=1800, bekle
